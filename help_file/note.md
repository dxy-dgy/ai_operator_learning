```bash
# change directory
cd

cp path_file1 path1
# print working directory
pwd
# list all the files
ls
# remove a file
rm file
# remove a directory
rm -r directory
# create a new file
touch file
# create a new directory
mkdir directory
# rename a new file
mv old_name new_name
# but if you rename a file to an already existed file, your action could just overwrite the old file, so this is a very dangerous action
```
